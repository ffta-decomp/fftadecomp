The build directory must contain `toolchain/agbcc`; compiled intermediates stay in a temporary directory and no ROM is written.

```sh
FFTA_RETAIL=/absolute/path/to/ffta-us.gba python3 - <<'PY'
import hashlib, json, os, pathlib, sys, tempfile
sys.path.insert(0, 'tools')
import build_rom

sha = lambda data: hashlib.sha256(data).hexdigest()
config = json.loads(pathlib.Path('config/gameplay-50k-build.json').read_text())
reference = pathlib.Path(os.environ['FFTA_RETAIL']).read_bytes()
if len(reference) != config['size'] or sha(reference) != config['sha256']:
    raise SystemExit('Wrong retail revision')
build_rom.BUILD_ROOT = pathlib.Path('build').resolve()
for module in config['modules']:
    with tempfile.TemporaryDirectory(prefix='campaign-check-') as tmp:
        work = pathlib.Path(tmp)
        build_rom.validate_source_mix(module)
        native = build_rom.compile_module(module, work, config['rom_base'], None, reference)
        build_rom.check_linked_function_entries(module, work, config['rom_base'])
    start, size = module['offset'], module['size']
    if len(native) != size or sha(native) != sha(reference[start:start + size]):
        raise SystemExit(f"Mismatch: {module['name']}")
    print(f"{module['name']}: {size} bytes, SHA-256 {sha(native)}")
PY
```
